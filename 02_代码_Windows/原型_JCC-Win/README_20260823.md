```powershell

powershell -File .\jcc.ps1 selftest

# 实时监控

# 全部命令
powershell -File .\jcc.ps1 help
# 写入类命令需追加 -Confirm；所有写入自动留痕 %TEMP%\jcc_writelog.csv
```
